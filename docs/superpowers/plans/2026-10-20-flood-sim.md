# Township Flood Simulator Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A static web app that simulates a 2-hour design storm over a synthetic township with a drainage network, lets the user toggle preset drainage interventions, and compares flood outcomes and cost against a baseline in a 3D view.

**Architecture:** Pure-TypeScript simulation core (`src/sim/`) — a virtual-pipes shallow-water surface grid coupled to a capacity-routed pipe network — with no DOM or three.js imports. `src/scenario/` builds the township, storms and interventions as data. A Web Worker runs the engine and streams depth frames to the main thread, where three.js (`src/render/`) animates them and plain-HTML panels (`src/ui/`) show metrics and charts.

**Tech Stack:** Vite 5, TypeScript 5 (strict), three.js 0.169, uPlot 1.6, Vitest 2, tsx (for the tuning script).

**Spec:** `docs/superpowers/specs/2026-10-20-flood-sim-design.md`

## Global Constraints

- `src/sim/` must never import `three`, `uplot`, or touch DOM APIs.
- Grid: 128 × 128 cells, 4 m per cell. Timestep `dt = 0.5 s`. Storm 2 h + 1 h recession. Frames every 30 sim-seconds.
- Manning `n = 0.013`. Flood thresholds: road `0.15 m`, home `0.30 m`. Water-balance tolerance `1%`.
- MSMA IDF coefficients must come from MSMA 2nd Edition Table 2.B1. Until a human copies them in, the code ships clearly flagged `provisional: true`, and the UI must say so.
- Every cost is an indicative placeholder and must be labelled as such in the UI.
- The demo story must emerge from the physics; never hard-code or script results.
- Determinism: identical inputs produce identical outputs (no `Math.random`, no wall-clock in the sim).
- Commit after every task. Windows dev machine: commands below work in PowerShell and Git Bash.

Create `src/sim/surface.ts`:

```ts
import type { Grid } from './types';

export const G = 9.81;
/** Per-step flux damping standing in for bed friction (assumption; shown in the UI). */
export const DAMPING = 0.995;

export interface SurfaceState {
  h: Float64Array; // water depth per cell, m
  fL: Float64Array; // outflow fluxes to each neighbour, m³/s
  fR: Float64Array;
  fB: Float64Array; // south (j - 1)
  fT: Float64Array; // north (j + 1)
}

export function createSurface(grid: Grid): SurfaceState {
  const n = grid.nx * grid.ny;
  return {
    h: Float64Array.from(grid.h0),
    fL: new Float64Array(n),
    fR: new Float64Array(n),
    fB: new Float64Array(n),
    fT: new Float64Array(n),
  };
}

/**
 * One step of the virtual-pipes shallow-water scheme (Mei et al. 2007).
 * Domain edges are walls. Volume is conserved up to float error.
 * NaN is deliberately preserved (not clamped) so the engine's validator can catch it.
 */
export function stepSurface(grid: Grid, s: SurfaceState, dt: number): void {
  const { nx, ny, dx, z } = grid;
  const { h, fL, fR, fB, fT } = s;
  const area = dx * dx;
  const k = dt * G * dx; // dt * g * A / l with A = dx², l = dx

  for (let j = 0; j < ny; j++) {
    for (let i = 0; i < nx; i++) {
      const c = j * nx + i;
      const H = z[c] + h[c];
      let l = i > 0 ? DAMPING * fL[c] + k * (H - z[c - 1] - h[c - 1]) : 0;
      let r = i < nx - 1 ? DAMPING * fR[c] + k * (H - z[c + 1] - h[c + 1]) : 0;
      let b = j > 0 ? DAMPING * fB[c] + k * (H - z[c - nx] - h[c - nx]) : 0;
      let t = j < ny - 1 ? DAMPING * fT[c] + k * (H - z[c + nx] - h[c + nx]) : 0;
      if (l < 0) l = 0;
      if (r < 0) r = 0;
      if (b < 0) b = 0;
      if (t < 0) t = 0;
      const out = (l + r + b + t) * dt;
      const vol = h[c] * area;
      if (out > vol) {
        const K = out > 0 ? vol / out : 0;
        l *= K;
        r *= K;
        b *= K;
        t *= K;
      }
      fL[c] = l;
      fR[c] = r;
      fB[c] = b;
      fT[c] = t;
    }
  }

  for (let j = 0; j < ny; j++) {
    for (let i = 0; i < nx; i++) {
      const c = j * nx + i;
      const inflow =
        (i > 0 ? fR[c - 1] : 0) +
        (i < nx - 1 ? fL[c + 1] : 0) +
        (j > 0 ? fT[c - nx] : 0) +
        (j < ny - 1 ? fB[c + nx] : 0);
      const outflow = fL[c] + fR[c] + fB[c] + fT[c];
      const next = h[c] + (dt * (inflow - outflow)) / area;
      h[c] = next < 0 ? 0 : next;
    }
  }
}
```

Create `src/sim/pipes.ts`:

```ts
import type { Network } from './types';

export const MANNING_N = 0.013;
export const MIN_SLOPE = 0.001;

/** Full-bore capacity of a circular pipe by Manning's equation (SI units), m³/s. */
export function manningFullCapacity(diameter: number, slope: number, n = MANNING_N): number {
  const A = (Math.PI * diameter * diameter) / 4;
  const R = diameter / 4;
  return (1 / n) * A * Math.pow(R, 2 / 3) * Math.sqrt(Math.max(slope, MIN_SLOPE));
}

export interface NetworkState {
  index: Map<string, number>;
  vol: Float64Array; // stored volume per node, m³
  vmax: Float64Array;
  from: Int32Array; // per pipe
  to: Int32Array;
  qFull: Float64Array;
  q: Float64Array; // flow this step, m³/s
  scratchOut: Float64Array;
  scratchIn: Float64Array;
}

export function createNetworkState(net: Network): NetworkState {
  const index = new Map(net.nodes.map((n, k) => [n.id, k] as const));
  const lookup = (id: string): number => {
    const k = index.get(id);
    if (k === undefined) throw new Error(`Unknown node ${id}`);
    return k;
  };
  return {
    index,
    vol: new Float64Array(net.nodes.length),
    vmax: Float64Array.from(net.nodes, (n) => n.area * n.maxDepth),
    from: Int32Array.from(net.pipes, (p) => lookup(p.from)),
    to: Int32Array.from(net.pipes, (p) => lookup(p.to)),
    qFull: Float64Array.from(net.pipes, (p) => manningFullCapacity(p.diameter, p.slope, p.n)),
    q: new Float64Array(net.pipes.length),
    scratchOut: new Float64Array(net.nodes.length),
    scratchIn: new Float64Array(net.nodes.length),
  };
}

/**
 * Capacity-based routing: each pipe carries min(full-bore capacity, water waiting upstream),
 * throttled when the downstream node lacks free storage. Mass-conservative.
 * A node can end above vmax only if its own outflow was throttled; coupling spills that excess.
 */
export function routePipes(st: NetworkState, dt: number): void {
  const { vol, vmax, from, to, qFull, q, scratchOut: out, scratchIn: inn } = st;
  const nP = q.length;

  out.fill(0);
  for (let p = 0; p < nP; p++) out[from[p]] += qFull[p];
  for (let p = 0; p < nP; p++) {
    const req = out[from[p]];
    const avail = vol[from[p]] / dt;
    q[p] = req > avail ? (qFull[p] * avail) / req : qFull[p];
  }

  out.fill(0);
  inn.fill(0);
  for (let p = 0; p < nP; p++) {
    out[from[p]] += q[p];
    inn[to[p]] += q[p];
  }
  for (let p = 0; p < nP; p++) {
    const d = to[p];
    const allowed = Math.max(0, (vmax[d] - vol[d]) / dt + out[d]);
    if (inn[d] > allowed) q[p] *= allowed / inn[d];
  }

  for (let p = 0; p < nP; p++) {
    vol[from[p]] -= q[p] * dt;
    vol[to[p]] += q[p] * dt;
  }
}
```

```ts
import type { Grid, RiverStage } from './types';
import type { SurfaceState } from './surface';

/** River water level at time t (linear interpolation, clamped at the ends). */
export function stageAt(river: RiverStage, t: number): number {
  const { times, levels } = river;
  if (t <= times[0]) return levels[0];
  for (let k = 1; k < times.length; k++) {
    if (t <= times[k]) {
      const f = (t - times[k - 1]) / (times[k] - times[k - 1]);
      return levels[k - 1] + f * (levels[k] - levels[k - 1]);
    }
  }
  return levels[levels.length - 1];
}

/** Forces river cells to the stage level. Returns the volumes added and removed (m³) for the water balance. */
export function resetRiver(grid: Grid, surf: SurfaceState, riverCells: Int32Array, level: number): { added: number; removed: number } {
  const area = grid.dx * grid.dx;
  let added = 0;
  let removed = 0;
  for (const c of riverCells) {
    const target = Math.max(0, level - grid.z[c]);
    const delta = (target - surf.h[c]) * area;
    if (delta > 0) added += delta;
    else removed -= delta;
    surf.h[c] = target;
  }
  return { added, removed };
}
```

```ts
import type { Grid, Network } from './types';
import type { NetworkState } from './pipes';
import { G, type SurfaceState } from './surface';

export const WEIR_C = 1.66; // SI weir coefficient for kerb/grate inlets (assumption)
export const ORIFICE_CD = 0.6; // outfall discharge coefficient (assumption)

/** Surface water enters nodes through inlets at the weir rate, limited by water present and free storage. */
export function captureInlets(grid: Grid, surf: SurfaceState, net: Network, st: NetworkState, dt: number): void {
  const area = grid.dx * grid.dx;
  for (let k = 0; k < net.nodes.length; k++) {
    const node = net.nodes[k];
    if (node.inletPerimeter <= 0) continue;
    const h = surf.h[node.cell];
    if (h <= 0) continue;
    const weir = WEIR_C * node.inletPerimeter * Math.pow(h, 1.5);
    const q = Math.min(weir, (h * area) / dt, Math.max(0, st.vmax[k] - st.vol[k]) / dt);
    const next = h - (q * dt) / area;
    surf.h[node.cell] = next < 0 ? 0 : next;
    st.vol[k] += q * dt;
  }
}

/** Water above a node's storage comes back out onto the street at that node's cell. */
export function spillOverflow(grid: Grid, surf: SurfaceState, net: Network, st: NetworkState): number {
  const area = grid.dx * grid.dx;
  let spilled = 0;
  for (let k = 0; k < net.nodes.length; k++) {
    const excess = st.vol[k] - st.vmax[k];
    if (excess > 0) {
      st.vol[k] = st.vmax[k];
      surf.h[net.nodes[k].cell] += excess / area;
      spilled += excess;
    }
  }
  return spilled;
}

/**
 * Orifice exchange between each outfall node and its river cell.
 * Positive = discharge to river; negative = river backflow into the network (blocked by a flap gate).
 */
export function exchangeOutfalls(
  grid: Grid,
  surf: SurfaceState,
  net: Network,
  st: NetworkState,
  riverLevel: number,
  dt: number,
): number {
  const area = grid.dx * grid.dx;
  let total = 0;
  for (const o of net.outfalls) {
    const k = st.index.get(o.node)!;
    const node = net.nodes[k];
    const head = node.invert + st.vol[k] / node.area;
    const dH = head - Math.max(riverLevel, node.invert);
    let q = 0;
    if (dH > 0) {
      q = Math.min(ORIFICE_CD * o.area * Math.sqrt(2 * G * dH), st.vol[k] / dt);
    } else if (dH < 0 && !o.flapGate) {
      q = -Math.min(ORIFICE_CD * o.area * Math.sqrt(-2 * G * dH), (surf.h[o.riverCell] * area) / dt);
    }
    st.vol[k] -= q * dt;
    surf.h[o.riverCell] += (q * dt) / area;
    total += q;
  }
  return total;
}

export function runPumps(grid: Grid, surf: SurfaceState, net: Network, st: NetworkState, dt: number): void {
  const area = grid.dx * grid.dx;
  for (const p of net.pumps) {
    const k = st.index.get(p.node)!;
    const q = Math.min(p.rate, st.vol[k] / dt);
    st.vol[k] -= q * dt;
    surf.h[p.riverCell] += (q * dt) / area;
  }
}
```

```ts
import { LandUse, type Frame, type Metrics, type Scenario } from './types';

export interface MetricsTracker {
  afterStep(h: Float64Array, outfallQ: number): void;
  onFrame(f: Frame): void;
  finish(stormEnd: number, balance: { rain: number; error: number }): Metrics;
}

export function createMetricsTracker(sc: Scenario): MetricsTracker {
  const { grid, thresholds, sagCell } = sc;
  const peak = new Float32Array(grid.nx * grid.ny);
  const roadCells: number[] = [];
  grid.land.forEach((l, c) => {
    if (l === LandUse.ROAD) roadCells.push(c);
  });
  let peakOutfall = 0;
  let lastFloodedT = 0;
  let floodedAtEnd = false;
  const sagT: number[] = [];
  const sagD: number[] = [];

  return {
    afterStep(h, outfallQ) {
      for (let c = 0; c < h.length; c++) if (h[c] > peak[c]) peak[c] = h[c];
      if (outfallQ > peakOutfall) peakOutfall = outfallQ;
    },
    onFrame(f) {
      sagT.push(f.t / 60);
      sagD.push(f.depth[sagCell]);
      floodedAtEnd = roadCells.some((c) => f.depth[c] >= thresholds.road);
      if (floodedAtEnd) lastFloodedT = f.t;
    },
    finish(stormEnd, balance) {
      const area = grid.dx * grid.dx;
      const homes = new Set<number>();
      grid.building.forEach((b, c) => {
        if (b >= 0 && peak[c] >= thresholds.home) homes.add(b);
      });
      return {
        peakDepth: peak,
        peakSagDepth: peak[sagCell],
        floodedRoadArea: roadCells.filter((c) => peak[c] >= thresholds.road).length * area,
        homesFlooded: homes.size,
        minutesToPassable: floodedAtEnd ? null : Math.max(0, lastFloodedT - stormEnd) / 60,
        peakOutfallFlow: peakOutfall,
        sagSeries: { t: sagT, depth: sagD },
        balance,
      };
    },
  };
}
```

```ts
import { captureInlets, exchangeOutfalls, runPumps, spillOverflow } from './coupling';
import { createMetricsTracker } from './metrics';
import { createNetworkState, routePipes, type NetworkState } from './pipes';
import { resetRiver, stageAt } from './river';
import { createSurface, stepSurface, type SurfaceState } from './surface';
import { LandUse, SimError, type Frame, type Scenario, type SimResult } from './types';

export interface EngineState {
  t: number;
  step: number;
  surf: SurfaceState;
  net: NetworkState;
  rainVol: number;
  infilVol: number;
  riverIn: number;
  riverOut: number;
  storage0: number;
  riverCells: Int32Array;
  landCells: Int32Array;
}

export function totalStorage(sc: Scenario, st: { surf: SurfaceState; net: NetworkState }): number {
  const area = sc.grid.dx * sc.grid.dx;
  let s = 0;
  for (const v of st.surf.h) s += v * area;
  for (const v of st.net.vol) s += v;
  return s;
}

export function createEngine(sc: Scenario): EngineState {
  const surf = createSurface(sc.grid);
  const net = createNetworkState(sc.network);
  const river: number[] = [];
  const land: number[] = [];
  sc.grid.land.forEach((l, c) => (l === LandUse.RIVER ? river : land).push(c));
  const riverCells = Int32Array.from(river);
  // Bring the river to its initial stage before measuring storage, so that isn't counted as inflow.
  resetRiver(sc.grid, surf, riverCells, stageAt(sc.river, 0));
  const st: EngineState = {
    t: 0,
    step: 0,
    surf,
    net,
    rainVol: 0,
    infilVol: 0,
    riverIn: 0,
    riverOut: 0,
    storage0: 0,
    riverCells,
    landCells: Int32Array.from(land),
  };
  st.storage0 = totalStorage(sc, st);
  return st;
}

function rainRate(sc: Scenario, t: number): number {
  const k = Math.floor(t / sc.storm.blockS);
  return k < sc.storm.intensity.length ? sc.storm.intensity[k] : 0;
}

/** Advances one timestep. Returns total outfall flow (m³/s, negative = backflow). */
export function stepEngine(sc: Scenario, st: EngineState): number {
  const { grid, dt } = sc;
  const area = grid.dx * grid.dx;
  const h = st.surf.h;
  const level = stageAt(sc.river, st.t);

  const r = resetRiver(grid, st.surf, st.riverCells, level);
  st.riverIn += r.added;
  st.riverOut += r.removed;

  const rain = rainRate(sc, st.t) * dt;
  for (const c of st.landCells) {
    const wet = h[c] + rain;
    const loss = Math.min(wet, grid.infil[c] * dt);
    h[c] = wet - loss;
    st.infilVol += loss * area;
  }
  st.rainVol += rain * area * st.landCells.length;

  stepSurface(grid, st.surf, dt);
  captureInlets(grid, st.surf, sc.network, st.net, dt);
  routePipes(st.net, dt);
  const outfallQ = exchangeOutfalls(grid, st.surf, sc.network, st.net, level, dt);
  runPumps(grid, st.surf, sc.network, st.net, dt);
  spillOverflow(grid, st.surf, sc.network, st.net);

  st.t += dt;
  st.step++;
  validate(st);
  return outfallQ;
}

function validate(st: EngineState): void {
  const h = st.surf.h;
  for (let c = 0; c < h.length; c++) {
    const v = h[c];
    if (!Number.isFinite(v) || v < 0) throw new SimError(`Invalid water depth ${v} at cell ${c}`, st.step);
  }
  const vol = st.net.vol;
  for (let k = 0; k < vol.length; k++) {
    if (!Number.isFinite(vol[k]) || vol[k] < -1e-9) throw new SimError(`Invalid node volume ${vol[k]} at node ${k}`, st.step);
  }
}

export function balanceError(sc: Scenario, st: EngineState): { rain: number; error: number } {
  const expected = st.storage0 + st.rainVol + st.riverIn - st.riverOut - st.infilVol;
  const error = Math.abs(totalStorage(sc, st) - expected) / Math.max(st.rainVol, 1e-9);
  return { rain: st.rainVol, error };
}

export function makeFrame(st: EngineState): Frame {
  const { vol, vmax, q, qFull } = st.net;
  return {
    t: st.t,
    depth: Float32Array.from(st.surf.h),
    nodeFill: Float32Array.from(vol, (v, k) => (vmax[k] > 0 ? v / vmax[k] : 0)),
    pipeLoad: Float32Array.from(q, (v, p) => (qFull[p] > 0 ? v / qFull[p] : 0)),
  };
}

export function runSimulation(sc: Scenario, onFrame?: (f: Frame) => void): SimResult {
  const st = createEngine(sc);
  const stormEnd = sc.storm.intensity.length * sc.storm.blockS;
  const nSteps = Math.round((stormEnd + sc.recessionS) / sc.dt);
  const frameEvery = Math.max(1, Math.round(sc.frameEveryS / sc.dt));
  const tracker = createMetricsTracker(sc);
  const frames: Frame[] = [];
  for (let s = 0; s < nSteps; s++) {
    const q = stepEngine(sc, st);
    tracker.afterStep(st.surf.h, q);
    if ((s + 1) % frameEvery === 0 || s === nSteps - 1) {
      const f = makeFrame(st);
      frames.push(f);
      tracker.onFrame(f);
      onFrame?.(f);
    }
  }
  return { frames, metrics: tracker.finish(stormEnd, balanceError(sc, st)) };
}
```

```ts
import type { Storm } from '../sim/types';

/** MSMA 2nd Ed. IDF form: i (mm/hr) = λ·T^κ / (d + θ)^η, with d in hours. */
export interface IdfCoefficients {
  station: string;
  lambda: number;
  kappa: number;
  theta: number;
  eta: number;
  provisional: boolean;
}

/**
 * PROVISIONAL coefficients of plausible Klang Valley magnitude, so the app runs.
 * Before the pitch: copy the coefficients for a Klang Valley station from
 * MSMA 2nd Edition (2012), Table 2.B1, set `station`, and set provisional: false.
 */
export const IDF: IdfCoefficients = {
  station: 'Provisional — replace from MSMA 2nd Ed. Table 2.B1',
  lambda: 60,
  kappa: 0.17,
  theta: 0.25,
  eta: 0.8,
  provisional: true,
};

export const STORM_DURATION_S = 7200;
export const BLOCK_S = 300;

export function idfIntensity(c: IdfCoefficients, returnPeriod: number, durationHours: number): number {
  return (c.lambda * Math.pow(returnPeriod, c.kappa)) / Math.pow(durationHours + c.theta, c.eta);
}

/** Alternating-block hyetograph (mm/hr per block), largest block at the centre. */
export function alternatingBlock(c: IdfCoefficients, returnPeriod: number, durationS: number, blockS: number): number[] {
  const n = Math.round(durationS / blockS);
  const increments: number[] = [];
  let prev = 0;
  for (let k = 1; k <= n; k++) {
    const dH = (k * blockS) / 3600;
    const cum = idfIntensity(c, returnPeriod, dH) * dH;
    increments.push(cum - prev);
    prev = cum;
  }
  increments.sort((a, b) => b - a);
  const out = new Array<number>(n).fill(0);
  const mid = Math.floor(n / 2);
  increments.forEach((d, k) => {
    const offset = k % 2 === 1 ? -(k + 1) / 2 : k / 2;
    out[mid + offset] = d / (blockS / 3600);
  });
  return out;
}

export function designStorm(returnPeriod: number, c: IdfCoefficients = IDF): Storm {
  return {
    id: `${returnPeriod}yr`,
    label: `${returnPeriod}-year ARI, 2 h`,
    returnPeriod,
    blockS: BLOCK_S,
    intensity: alternatingBlock(c, returnPeriod, STORM_DURATION_S, BLOCK_S).map((mmhr) => mmhr / 3.6e6),
    provisional: c.provisional,
  };
}

export const STORMS = { '10yr': designStorm(10), '100yr': designStorm(100) };
export type StormId = keyof typeof STORMS;
```

```ts
import { LandUse, type Grid } from '../sim/types';
import { BOULEVARD_I, BUND, GRID, INFILTRATION_MMHR, MMHR, POND, RIVER_ROWS, SAG, TERRAIN } from './config';

export function cellAt(x: number, y: number): number {
  return Math.floor(y / GRID.dx) * GRID.nx + Math.floor(x / GRID.dx);
}

/** Infiltration (m/s) indexed by LandUse code. */
export const INFIL_BY_LAND: number[] = [
  INFILTRATION_MMHR.ROOF,
  INFILTRATION_MMHR.ROAD,
  INFILTRATION_MMHR.LAWN,
  INFILTRATION_MMHR.PARK,
  INFILTRATION_MMHR.POND,
  INFILTRATION_MMHR.RIVER,
].map((v) => v * MMHR);

function bowl(x: number, y: number, cx: number, cy: number, r: number, depth: number): number {
  const d = Math.hypot(x - cx, y - cy);
  return d < r ? depth * (1 - (d / r) ** 2) : 0;
}

/** Deterministic synthetic township: hill (north) → housing grid → sag (SW) + pond (SE) → river (south). */
export function buildTerrain(): Grid {
  const { nx, ny, dx } = GRID;
  const n = nx * ny;
  const z = new Float32Array(n);
  const land = new Uint8Array(n);
  const infil = new Float32Array(n);
  const building = new Int32Array(n).fill(-1);
  const h0 = new Float32Array(n);

  for (let j = 0; j < ny; j++) {
    for (let i = 0; i < nx; i++) {
      const c = j * nx + i;
      const x = (i + 0.5) * dx;
      const y = (j + 0.5) * dx;
      if (j < RIVER_ROWS) {
        z[c] = 0;
        land[c] = LandUse.RIVER;
        continue;
      }
      let zc = TERRAIN.baseZ + TERRAIN.slope * y + Math.max(0, y - TERRAIN.hillStartY) * TERRAIN.hillSlope;
      zc -= bowl(x, y, SAG.x, SAG.y, SAG.r, SAG.depth);
      zc -= bowl(x, y, POND.x, POND.y, POND.r, POND.depth);
      const pondDist = Math.hypot(x - POND.x, y - POND.y);

      let lc: number = LandUse.LAWN;
      if (pondDist < POND.r) lc = LandUse.POND;
      else if (pondDist < POND.r + POND.parkMargin || y > TERRAIN.hillStartY) lc = LandUse.PARK;
      else if (i % 10 === 0 || j % 10 === 0 || Math.abs(i - BOULEVARD_I) <= 1) {
        lc = LandUse.ROAD;
        zc -= TERRAIN.roadDrop;
      } else if (i % 10 >= 2 && i % 10 <= 8 && j % 10 >= 2 && j % 10 <= 8) {
        lc = LandUse.ROOF;
        building[c] = Math.floor(i / 10) + Math.floor(j / 10) * 100;
      }

      if (j < RIVER_ROWS + 2 && x >= BUND.xMin && x <= BUND.xMax) {
        zc = Math.max(zc, BUND.z);
        lc = LandUse.LAWN;
        building[c] = -1;
      }

      z[c] = zc;
      land[c] = lc;
      if (lc === LandUse.POND) h0[c] = Math.max(0, POND.initialLevel - zc);
    }
  }
  for (let c = 0; c < n; c++) infil[c] = INFIL_BY_LAND[land[c]];
  return { nx, ny, dx, z, land, infil, building, h0 };
}
```

```ts
import { MANNING_N } from '../sim/pipes';
import type { DrainNode, Grid, Network, Pipe } from '../sim/types';
import { COVER, MANHOLE, PIPE_D } from './config';
import { cellAt } from './terrain';

export const OUTFALL_RIVER_POINT = { x: 82, y: 10 };

export function makeNode(
  grid: Grid,
  id: string,
  x: number,
  y: number,
  opts: Partial<Pick<DrainNode, 'invert' | 'area' | 'maxDepth' | 'inletPerimeter'>> = {},
): DrainNode {
  const cell = cellAt(x, y);
  const ground = grid.z[cell];
  const invert = opts.invert ?? ground - COVER;
  return {
    id,
    x,
    y,
    cell,
    invert,
    area: opts.area ?? MANHOLE.area,
    maxDepth: opts.maxDepth ?? ground - invert,
    inletPerimeter: opts.inletPerimeter ?? MANHOLE.inletPerimeter,
  };
}

export function makePipe(a: DrainNode, b: DrainNode, diameter: number): Pipe {
  const length = Math.hypot(a.x - b.x, a.y - b.y);
  return { id: `${a.id}-${b.id}`, from: a.id, to: b.id, diameter, length, slope: (a.invert - b.invert) / length, n: MANNING_N };
}
```

```ts
  to: string;

export const CUSTOM_PIPE_RM_PER_M_PER_M_DIA = 5_000;

let cachedBase: Township | null = null;

export function baseTownship(): Township {
  if (!cachedBase) {
    const grid = buildTerrain();
    cachedBase = { grid, network: buildNetwork(grid) };

export function buildTownship(ids: readonly InterventionId[], custom: readonly CustomPipe[] = []): Township {
  const t = applyInterventions(baseTownship(), ids);
  custom.forEach((cp, k) => {
    const pipe = makePipe(findNode(t, cp.from), findNode(t, cp.to), cp.diameter);
    t.network.pipes.push({ ...pipe, id: `custom-${k}` });
  });
  return t;
}

export function customPipeCost(t: Township, cp: CustomPipe): number {
  const a = findNode(t, cp.from);
  const b = findNode(t, cp.to);
  return Math.hypot(a.x - b.x, a.y - b.y) * cp.diameter * CUSTOM_PIPE_RM_PER_M_PER_M_DIA;
}

    frameEveryS: SIM.frameEveryS,
```

```ts
import { scenarioKey, type CustomPipe } from '../scenario/index';
import type { InterventionId } from '../scenario/interventions';
import type { StormId } from '../scenario/storms';
import type { Frame, Metrics } from '../sim/types';
import type { RunRequest, WorkerMessage } from '../worker/protocol';

export const MAX_CACHE = 6;

export interface RunResult {
  frames: Frame[];
  metrics: Metrics;
}

export interface WorkerLike {
  postMessage(m: RunRequest): void;
  onmessage: ((e: MessageEvent<WorkerMessage>) => void) | null;
  onerror: ((e: ErrorEvent) => void) | null;
  terminate(): void;
}

export class SimRunError extends Error {
  constructor(
    message: string,
    public readonly step: number,
  ) {
    super(message);
    this.name = 'SimRunError';
  }
}

/** Runs one simulation at a time on a worker; caches results (LRU) by scenario key. */
export class SimRunner {
  private cache = new Map<string, RunResult>();
  private queue: Promise<unknown> = Promise.resolve();
  private worker: WorkerLike;

  constructor(private readonly createWorker: () => WorkerLike) {
    this.worker = createWorker();
  }

  run(ids: InterventionId[], stormId: StormId, onFrame?: (f: Frame) => void, custom: CustomPipe[] = []): Promise<RunResult> {
    const job = this.queue.then(() => this.execute(ids, stormId, onFrame, custom));
    this.queue = job.catch(() => undefined);
    return job;
  }

  isCached(ids: InterventionId[], stormId: StormId, custom: CustomPipe[] = []): boolean {
    return this.cache.has(scenarioKey(ids, stormId, custom));
  }

  private execute(ids: InterventionId[], stormId: StormId, onFrame: ((f: Frame) => void) | undefined, custom: CustomPipe[]): Promise<RunResult> {
    const key = scenarioKey(ids, stormId, custom);
    const hit = this.cache.get(key);
    if (hit) {
      this.cache.delete(key);
      this.cache.set(key, hit);
      hit.frames.forEach((f) => onFrame?.(f));
      return Promise.resolve(hit);
    }
    return new Promise<RunResult>((resolve, reject) => {
      const frames: Frame[] = [];
      this.worker.onmessage = (e) => {
        const m = e.data;
        if (m.key !== key) return;
        if (m.type === 'frame') {
          frames.push(m.frame);
          onFrame?.(m.frame);
        } else if (m.type === 'done') {
          const result = { frames, metrics: m.metrics };
          this.remember(key, result);
          resolve(result);
        } else {
          reject(new SimRunError(m.message, m.step));
        }
      };
      this.worker.onerror = (e) => {
        this.worker.terminate();
        this.worker = this.createWorker();
        reject(new SimRunError(e.message || 'Simulation worker crashed', -1));
      };
      this.worker.postMessage({ type: 'run', key, interventions: [...ids].sort(), stormId, custom });
    });
  }

  private remember(key: string, result: RunResult): void {
    this.cache.set(key, result);
    while (this.cache.size > MAX_CACHE) this.cache.delete(this.cache.keys().next().value!);
  }
}
```

```ts
import * as THREE from 'three';
import type { Grid } from '../sim/types';
import { worldX, worldZ } from './coords';

/** One vertex per cell centre, two upward-facing triangles per quad. y and colour are filled by the caller. */
export function createGridGeometry(grid: Grid): THREE.BufferGeometry {
  const { nx, ny, dx } = grid;
  const n = nx * ny;
  const pos = new Float32Array(n * 3);
  for (let j = 0; j < ny; j++) {
    for (let i = 0; i < nx; i++) {
      const c = j * nx + i;
      pos[3 * c] = worldX(grid, (i + 0.5) * dx);
      pos[3 * c + 2] = worldZ(grid, (j + 0.5) * dx);
    }
  }
  const index: number[] = [];
  for (let j = 0; j < ny - 1; j++) {
    for (let i = 0; i < nx - 1; i++) {
      const a = j * nx + i;
      const b = a + 1;
      const d = a + nx;
      const e = d + 1;
      index.push(a, b, d, b, e, d);
    }
  }
  const geom = new THREE.BufferGeometry();
  geom.setIndex(index);
  geom.setAttribute('position', new THREE.BufferAttribute(pos, 3));
  geom.setAttribute('color', new THREE.BufferAttribute(new Float32Array(n * 3), 3));
  return geom;
}
```

```ts
const HEIGHT = 8; // world units (not exaggerated)

/** One box per building id, sized to its footprint. */
export function createBuildings(grid: Grid): THREE.InstancedMesh {
  const boxes = new Map<number, { i0: number; i1: number; j0: number; j1: number; zMin: number }>();
  for (let j = 0; j < grid.ny; j++) {
    for (let i = 0; i < grid.nx; i++) {
      const c = j * grid.nx + i;
      const id = grid.building[c];
      if (id < 0) continue;
      const b = boxes.get(id);
      if (!b) boxes.set(id, { i0: i, i1: i, j0: j, j1: j, zMin: grid.z[c] });
      else {
        b.i0 = Math.min(b.i0, i);
        b.i1 = Math.max(b.i1, i);
        b.j0 = Math.min(b.j0, j);
        b.j1 = Math.max(b.j1, j);
        b.zMin = Math.min(b.zMin, grid.z[c]);
      }
    }
  }
  const mesh = new THREE.InstancedMesh(
    new THREE.BoxGeometry(1, 1, 1),
    new THREE.MeshLambertMaterial({ color: '#efe6d8' }),
    boxes.size,
  );
  const m = new THREE.Matrix4();
  let k = 0;
  for (const b of boxes.values()) {
    const w = (b.i1 - b.i0 + 1) * grid.dx;
    const d = (b.j1 - b.j0 + 1) * grid.dx;
    const x = worldX(grid, b.i0 * grid.dx + w / 2);
    const z = worldZ(grid, b.j0 * grid.dx + d / 2);
    m.compose(new THREE.Vector3(x, worldY(b.zMin) + HEIGHT / 2, z), new THREE.Quaternion(), new THREE.Vector3(w, HEIGHT, d));
    mesh.setMatrixAt(k++, m);
  }
  mesh.instanceMatrix.needsUpdate = true;
  return mesh;
}
```

```ts
import type { Grid } from '../sim/types';

const SHALLOW = new THREE.Color('#9fd3ff');
const DEEP = new THREE.Color('#08306b');
const WET = 0.01; // m
const FULL_COLOUR_DEPTH = 1.0; // m

export class WaterSurface {
  readonly mesh: THREE.Mesh;
  private readonly geom: THREE.BufferGeometry;
  private readonly tmp = new THREE.Color();

  constructor(private readonly grid: Grid) {
    this.geom = createGridGeometry(grid);
    this.mesh = new THREE.Mesh(
      this.geom,
      new THREE.MeshPhongMaterial({ vertexColors: true, transparent: true, opacity: 0.82, shininess: 80 }),
    );
    this.update(grid.h0);
  }

  update(depth: ArrayLike<number>): void {
    const pos = this.geom.getAttribute('position') as THREE.BufferAttribute;
    const col = this.geom.getAttribute('color') as THREE.BufferAttribute;
    const { z } = this.grid;
    for (let c = 0; c < depth.length; c++) {
      const d = depth[c];
      pos.setY(c, d > WET ? worldY(z[c] + d) + 0.05 : worldY(z[c]) - 1);
      this.tmp.copy(SHALLOW).lerp(DEEP, Math.min(1, d / FULL_COLOUR_DEPTH));
      col.setXYZ(c, this.tmp.r, this.tmp.g, this.tmp.b);
    }
    pos.needsUpdate = true;
    col.needsUpdate = true;
    this.geom.computeVertexNormals();
  }

  dispose(): void {
    this.geom.dispose();
    (this.mesh.material as THREE.Material).dispose();
  }
}
```

```ts
      mesh.quaternion.setFromUnitVectors(up, dir.normalize());

    for (const n of net.nodes) {
      const mesh = new THREE.Mesh(new THREE.SphereGeometry(1.8, 12, 8), new THREE.MeshBasicMaterial({ color: NODE_OK.clone() }));
      mesh.position.copy(at(n));
      mesh.userData.nodeId = n.id;
      this.nodeMeshes.push(mesh);
      this.group.add(mesh);
    }
  }

  update(frame: Frame | null, time: number): void {
    this.pipeMeshes.forEach((m, p) => loadColor(frame ? frame.pipeLoad[p] : 0, (m.material as THREE.MeshBasicMaterial).color));
    this.nodeMeshes.forEach((m, k) => {
      const surcharged = frame !== null && frame.nodeFill[k] >= 0.999;
      (m.material as THREE.MeshBasicMaterial).color.copy(surcharged ? RED : NODE_OK);
      m.scale.setScalar(surcharged ? 1 + 0.35 * Math.sin(time * 6) : 1);
    });
  }

  dispose(): void {
    for (const m of [...this.pipeMeshes, ...this.nodeMeshes]) {
      m.geometry.dispose();
      (m.material as THREE.Material).dispose();
    }
  }
```

```ts
import * as THREE from 'three';
import { OrbitControls } from 'three/addons/controls/OrbitControls.js';
import type { Township } from '../scenario/interventions';
import type { Frame } from '../sim/types';
import { createBuildings } from './buildings';
import { NetworkView } from './network';
import { createTerrainMesh } from './terrain';
import { WaterSurface } from './water';

export interface SceneHandle {
  canvas: HTMLCanvasElement;
  setTownship(t: Township): void;
  showFrame(f: Frame | null): void;
  /** ndc in [-1, 1]; returns the id of the drain node under the pointer, if any. */
  pickNode(ndcX: number, ndcY: number): string | null;
}

export function createScene(container: HTMLElement): SceneHandle {
  const renderer = new THREE.WebGLRenderer({ antialias: true });
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  container.appendChild(renderer.domElement);

  const scene = new THREE.Scene();
  scene.background = new THREE.Color('#cfe3f3');
  const camera = new THREE.PerspectiveCamera(50, 1, 1, 5000);
  camera.position.set(-120, 380, 520);
  const controls = new OrbitControls(camera, renderer.domElement);
  controls.target.set(0, 20, 0);
  controls.enableDamping = true;

  scene.add(new THREE.HemisphereLight(0xffffff, 0x445544, 1.2));
  const sun = new THREE.DirectionalLight(0xffffff, 1.5);
  sun.position.set(200, 400, 100);
  scene.add(sun);

  let township: Township | null = null;
  let terrain: THREE.Mesh | null = null;
  let buildings: THREE.InstancedMesh | null = null;
  let water: WaterSurface | null = null;
  let network: NetworkView | null = null;
  let frame: Frame | null = null;

  const resize = () => {
    const { clientWidth: w, clientHeight: h } = container;
    renderer.setSize(w, h);
    camera.aspect = w / Math.max(1, h);
    camera.updateProjectionMatrix();
  };
  new ResizeObserver(resize).observe(container);
  resize();

  const clock = new THREE.Clock();
  renderer.setAnimationLoop(() => {
    controls.update();
    network?.update(frame, clock.getElapsedTime());
    renderer.render(scene, camera);
  });

  const raycaster = new THREE.Raycaster();

  return {
    canvas: renderer.domElement,
    setTownship(t) {
      for (const obj of [terrain, buildings]) {
        if (!obj) continue;
        scene.remove(obj);
        obj.geometry.dispose();
      }
      if (water) {
        scene.remove(water.mesh);
        water.dispose();
      }
      if (network) {
        scene.remove(network.group);
        network.dispose();
      }
      township = t;
      terrain = createTerrainMesh(t.grid);
      buildings = createBuildings(t.grid);
      water = new WaterSurface(t.grid);
      network = new NetworkView(t.grid, t.network);
      scene.add(terrain, buildings, water.mesh, network.group);
      frame = null;
    },
    showFrame(f) {
      frame = f;
      if (water && township) water.update(f ? f.depth : township.grid.h0);
    },
    pickNode(ndcX, ndcY) {
      if (!network) return null;
      raycaster.setFromCamera(new THREE.Vector2(ndcX, ndcY), camera);
      const hit = raycaster.intersectObjects(network.nodeMeshes, false)[0];
      return hit ? (hit.object.userData.nodeId as string) : null;
    },
  };
}
```

```ts
export type Tone = 'better' | 'worse' | 'same';

export function describeChange(base: number, cur: number, digits: number, lowerIsBetter = true): { text: string; tone: Tone } {
  const d = cur - base;
  if (Math.abs(d) < 10 ** -digits / 2) return { text: 'no change', tone: 'same' };
  const sign = d > 0 ? '+' : '';
  const pct = base !== 0 ? ` (${sign}${Math.round((d / base) * 100)}%)` : '';
  const improved = lowerIsBetter ? d < 0 : d > 0;
  return { text: `${sign}${d.toFixed(digits)}${pct}`, tone: improved ? 'better' : 'worse' };
}

export function formatRM(n: number): string {
  return n === 0 ? 'RM 0' : `RM ${(n / 1e6).toFixed(2)}M`;
}
```

Create `src/ui/player.ts`:

```ts
import type { Frame } from '../sim/types';

/** Plays frames at 10 fps as they arrive; the scrubber seeks; the button pauses/resumes. */
export class Player {
  private frames: Frame[] = [];
  private idx = 0;
  private playing = true;

  constructor(
    private readonly scrubber: HTMLInputElement,
    private readonly clock: HTMLElement,
    private readonly playBtn: HTMLButtonElement,
    private readonly show: (f: Frame) => void,
  ) {
    scrubber.addEventListener('input', () => {
      this.setPlaying(false);
      this.seek(Number(scrubber.value));
    });
    playBtn.addEventListener('click', () => {
      if (!this.playing && this.idx >= this.frames.length - 1) this.idx = 0;
      this.setPlaying(!this.playing);
    });
    setInterval(() => this.tick(), 100);
  }

  reset(): void {
    this.frames = [];
    this.idx = 0;
    this.scrubber.max = '0';
    this.scrubber.value = '0';
    this.setPlaying(true);
  }

  push(f: Frame): void {
    this.frames.push(f);
    this.scrubber.max = String(this.frames.length - 1);
  }

  private tick(): void {
    if (this.playing && this.idx < this.frames.length) {
      this.seek(this.idx);
      this.idx++;
    }
  }

  private seek(i: number): void {
    const f = this.frames[i];
    if (!f) return;
    this.idx = i;
    this.scrubber.value = String(i);
    this.clock.textContent = `${Math.round(f.t / 60)} min`;
    this.show(f);
  }

  private setPlaying(p: boolean): void {
    this.playing = p;
    this.playBtn.textContent = p ? 'Pause' : 'Play';
  }
}
```

```ts
import { buildScenario } from '../src/scenario/index';
import { INTERVENTIONS, totalCost, type InterventionId } from '../src/scenario/interventions';
import { runSimulation } from '../src/sim/engine';

const combos: Array<[string, InterventionId[]]> = [
  ['baseline', []],
  ...INTERVENTIONS.map((iv, k): [string, InterventionId[]] => [`#${k + 1} ${iv.name}`, [iv.id]]),
  ['#3 + #5 tank + flap', ['detention-tank', 'flap-gate']],
];

const rows = combos.map(([name, ids]) => {
  const t0 = performance.now();
  const m = runSimulation(buildScenario(ids, '100yr')).metrics;
  return {
    scenario: name,
    'sag peak (m)': m.peakSagDepth.toFixed(2),
    homes: m.homesFlooded,
    'road area (m²)': m.floodedRoadArea,
    'passable (min)': m.minutesToPassable === null ? '> 60' : m.minutesToPassable.toFixed(0),
    'cost (RM M)': (totalCost(ids) / 1e6).toFixed(2),
    'balance %': (m.balance.error * 100).toFixed(3),
    'run (s)': ((performance.now() - t0) / 1000).toFixed(1),
  };
});
console.table(rows);
```
